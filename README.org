# HTML parse
from bs4 import BeautifulSoup

# Load the HTML file
with open("/mnt/e/code/html/my-website/index.html", "r", encoding="utf-8") as file:
    soup = BeautifulSoup(file, "html.parser")

# Extract relevant information
elements = []
for el in soup.find_all(True):  # Finds all tags
    tag_name = el.name
    inline_style = el.get("style", "")
    class_name = el.get("class", [])
    elements.append({
        "tag": tag_name,
        "classes": class_name,
        "inline_style": inline_style,
    })

print(elements)  # List of elements with styles and classes

# CSS Parse
import cssutils

# Parse the CSS file
css_map = {}
with open("/mnt/e/code/html/my-website/src/index.css", "r", encoding="utf-8") as css_file:
    stylesheet = cssutils.parseString(css_file.read())

    for rule in stylesheet:
        if rule.type == rule.STYLE_RULE:
            selector = rule.selectorText
            styles = rule.style.cssText
            css_map[selector] = styles

print(css_map)  # Dictionary mapping selectors to styles

def extract_and_process_styles():
    """Extract HTML elements, parse CSS, and combine styles in one seamless process"""

    # 1. Parse HTML
    from bs4 import BeautifulSoup

    # Load the HTML file
    with open("/mnt/e/code/html/my-website/index.html", "r", encoding="utf-8") as file:
        soup = BeautifulSoup(file, "html.parser")

    # Extract relevant information
    elements = []
    for el in soup.find_all(True):  # Finds all tags
        tag_name = el.name
        inline_style = el.get("style", "")
        class_name = el.get("class", [])
        elements.append({
            "tag": tag_name,
            "classes": class_name,
            "inline_style": inline_style,
        })

    # 2. Parse CSS
    import cssutils

    # Parse the CSS file
    css_map = {}
    with open("/mnt/e/code/html/my-website/src/index.css", "r", encoding="utf-8") as css_file:
        stylesheet = cssutils.parseString(css_file.read())

        for rule in stylesheet:
            if rule.type == rule.STYLE_RULE:
                selector = rule.selectorText
                styles = rule.style.cssText
                css_map[selector] = styles

    # 3. Match Elements to Their Styles
    def resolve_styles(element, css_map):
        # Get inline styles
        inline_style = element["inline_style"]

        # Resolve class styles
        class_styles = []
        for class_name in element["classes"]:
            selector = f".{class_name}"
            if selector in css_map:
                class_styles.append(css_map[selector])

        # Combine all styles
        combined_styles = "; ".join([inline_style] + class_styles)
        return combined_styles

    # Process each element
    for element in elements:
        resolved_styles = resolve_styles(element, css_map)
        element["resolved_styles"] = resolved_styles

    return elements

# Execute the combined process

from typing import Dict, List, Optional
from bs4 import BeautifulSoup
import cssutils
import json

class TailwindJSONProcessor:
    def __init__(self, tailwind_json_path: str):
        """
        Initialize processor with tailwind_classes.json path

        Parameters:
            tailwind_json_path: str - Path to exported tailwind_classes.json
        """
        self.tailwind_config = self._load_tailwind_json(tailwind_json_path)
        cssutils.log.setLevel(30)  # Suppress warnings

    def _load_tailwind_json(self, json_path: str) -> Dict:
        """Load and parse tailwind_classes.json configuration"""
        try:
            with open(json_path, 'r') as f:
                config = json.load(f)
                print(f"Successfully loaded Tailwind configuration with {len(config)} entries")
                return config
        except FileNotFoundError:
            raise FileNotFoundError(
                f"tailwind_classes.json not found at {json_path}. "
                "Generate it using: npx tailwindcss-cli build --json > tailwind_classes.json"
            )
        except json.JSONDecodeError:
            raise ValueError("Invalid JSON in tailwind_classes.json")

    def process_files(self, html_path: str, css_path: str) -> List[dict]:
        """
        Process HTML and CSS files with Tailwind mapping

        Parameters:
            html_path: str - e.g., "/mnt/e/code/html/my-website/index.html"
            css_path: str - e.g., "/mnt/e/code/html/my-website/src/index.css"
        """
        # Parse HTML
        with open(html_path, "r", encoding="utf-8") as file:
            soup = BeautifulSoup(file, "html.parser")

        # Extract elements
        elements = []
        for el in soup.find_all(True):
            elements.append({
                "tag": el.name,
                "classes": el.get("class", []),
                "inline_style": el.get("style", ""),
                "original_html": str(el)
            })

        # Parse CSS
        css_map = self._parse_css_file(css_path)

        # Process elements with Tailwind mapping
        return self._process_elements(elements, css_map)

    def _parse_css_file(self, css_path: str) -> Dict[str, str]:
        """Parse CSS file and create selector-to-styles mapping"""
        css_map = {}
        with open(css_path, "r", encoding="utf-8") as css_file:
            stylesheet = cssutils.parseString(css_file.read())
            for rule in stylesheet:
                if rule.type == rule.STYLE_RULE:
                    selector = rule.selectorText
                    styles = rule.style.cssText
                    css_map[selector] = styles
        return css_map

    def _map_to_tailwind(self, css_property: str, css_value: str) -> Optional[str]:
        """Map CSS properties to Tailwind classes using tailwind_classes.json"""
        # First, normalize the CSS value
        css_value = css_value.strip().lower()

        # Check direct mapping in tailwind_classes.json
        potential_class = f"{css_property}-{css_value}"
        if potential_class in self.tailwind_config:
            return self.tailwind_config[potential_class]

        # Handle special cases
        if css_property in ['margin', 'padding', 'width', 'height']:
            return self._map_spacing_property(css_property, css_value)
        elif css_property in ['color', 'background-color']:
            return self._map_color_property(css_property, css_value)
        elif css_property == 'font-size':
            return self._map_font_size(css_value)

        return None

    def _map_spacing_property(self, property_type: str, value: str) -> Optional[str]:
        """Map spacing properties using Tailwind's spacing scale"""
        prefix_map = {
            'margin': 'm',
            'padding': 'p',
            'width': 'w',
            'height': 'h'
        }

        # Extract numeric value and unit
        import re
        match = re.match(r'^(-?\d*\.?\d+)(rem|px|em)?$', value)
        if not match:
            return None

        number, unit = match.groups()
        pixels = float(number)
        if unit == 'rem':
            pixels *= 16

        # Find closest Tailwind spacing value
        prefix = prefix_map.get(property_type, '')
        for tw_class, tw_value in self.tailwind_config.get('theme', {}).get('spacing', {}).items():
            if abs(float(tw_value.replace('px', '')) - pixels) < 0.1:
                return f"{prefix}-{tw_class}"

        return None

    def _map_color_property(self, property_type: str, value: str) -> Optional[str]:
        """Map color properties using Tailwind's color palette"""
        prefix = 'text' if property_type == 'color' else 'bg'

        # Handle hex colors
        if value.startswith('#'):
            for color_name, color_value in self.tailwind_config.get('theme', {}).get('colors', {}).items():
                if isinstance(color_value, dict):
                    for shade, hex_value in color_value.items():
                        if hex_value.lower() == value.lower():
                            return f"{prefix}-{color_name}-{shade}"
                elif color_value.lower() == value.lower():
                    return f"{prefix}-{color_name}"

        return None

    def _process_elements(self, elements: List[dict], css_map: Dict[str, str]) -> List[dict]:
        """Process elements with Tailwind class mapping"""
        for element in elements:
            # Combine styles
            styles = self._combine_styles(element, css_map)
            element["resolved_styles"] = styles

            # Convert to Tailwind classes
            tailwind_classes = []
            for property_value in styles.split(';'):
                if ':' in property_value:
                    prop, value = property_value.split(':', 1)
                    tw_class = self._map_to_tailwind(prop.strip(), value.strip())
                    if tw_class:
                        tailwind_classes.append(tw_class)

            element["tailwind_classes"] = ' '.join(tailwind_classes)

        return elements

    def _combine_styles(self, element: dict, css_map: Dict[str, str]) -> str:
        """Combine inline and CSS styles"""
        styles = []
        if element["inline_style"]:
            styles.append(element["inline_style"])

        for class_name in element["classes"]:
            selector = f".{class_name}"
            if selector in css_map:
                styles.append(css_map[selector])

        return '; '.join(filter(None, styles))

def main():
    """Main execution function"""
    # Initialize processor with tailwind_classes.json
    processor = TailwindJSONProcessor("tailwind_classes.json")

    # Process files with exact paths
    html_path = "/mnt/e/code/html/my-website/index.html"
    css_path = "/mnt/e/code/html/my-website/src/index.css"

    try:
        processed_elements = processor.process_files(html_path, css_path)

        # Output results
        for element in processed_elements:
            print(f"\nProcessed Element:")
            print(f"Tag: {element['tag']}")
            print(f"Original Classes: {element['classes']}")
            print(f"Resolved Styles: {element['resolved_styles']}")
            print(f"Tailwind Classes: {element.get('tailwind_classes', '')}")

    except Exception as e:
        print(f"Error processing styles: {str(e)}")

if __name__ == "__main__":
    main()
